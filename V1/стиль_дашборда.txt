let AC=null;
function ensureAC(){if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)()}catch(e){}}}
function beep(f){
  if(!S.sound)return;ensureAC();if(!AC)return;
  const o=AC.createOscillator(),g=AC.createGain();
  o.frequency.value=f;o.type='sine';
  g.gain.setValueAtTime(0.0001,AC.currentTime);
  g.gain.exponentialRampToValueAtTime(0.12,AC.currentTime+0.02);
  g.gain.exponentialRampToValueAtTime(0.0001,AC.currentTime+0.16);
  o.connect(g).connect(AC.destination);o.start();o.stop(AC.currentTime+0.18);
}
